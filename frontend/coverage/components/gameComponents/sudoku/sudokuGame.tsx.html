const checkSolution = (
  currentGrid: SudokuCell[][],
  solutionString: string
): boolean => {
  for (let r = 0; r < 9; r++) {
    for (let c = 0; c < 9; c++) {
      const expectedValue = parseInt(solutionString[r * 9 + c]);
      if (
        !currentGrid[r][c].value ||
        currentGrid[r][c].value !== expectedValue
      ) {
        return false;
      }
    }
  }
  return true;
};
